find_package(Torch REQUIRED)
set(KITTEN_BUILD_CLI OFF CACHE BOOL "Build normalizer CLI" FORCE)
add_subdirectory(${PROJECT_SOURCE_DIR}/vendor/kitten-text-processing ${CMAKE_CURRENT_BINARY_DIR}/text-processing)
add_library(kitten-text STATIC text.cpp)
target_link_libraries(kitten-text PUBLIC KittenTextProcessing::normalizer)
target_compile_features(kitten-text PUBLIC cxx_std_17)
add_library(kitten-decoder STATIC decoder.cpp)
target_include_directories(kitten-decoder PRIVATE ${PROJECT_SOURCE_DIR}/vendor/nlohmann)
target_link_libraries(kitten-decoder PUBLIC ${TORCH_LIBRARIES})
target_compile_features(kitten-decoder PUBLIC cxx_std_17)
add_library(kitten-engine STATIC engine.cpp assets.cpp)
target_include_directories(kitten-engine PUBLIC ${PROJECT_SOURCE_DIR}/vendor/nlohmann)
target_link_libraries(kitten-engine PUBLIC llama-common llama kitten-text kitten-decoder)
add_executable(kitten-tts main.cpp)
target_include_directories(kitten-tts PRIVATE ${PROJECT_SOURCE_DIR}/vendor/nlohmann)
target_link_libraries(kitten-tts PRIVATE kitten-engine)
if(KITTEN_DATA_DIR)
    set(kitten_data "${KITTEN_DATA_DIR}")
else()
    set(kitten_data "${CMAKE_CURRENT_BINARY_DIR}/text-processing/data")
endif()
target_compile_definitions(kitten-engine PUBLIC KITTEN_DEFAULT_DATA="${kitten_data}")

set(KITTEN_REFERENCE_PACKAGE "" CACHE PATH "KittenTTS Python checkout for differential tests")
set(KITTEN_TEST_PYTHON "python3" CACHE FILEPATH "Python with KittenTTS2 conversion dependencies")
if(BUILD_TESTING AND KITTEN_REFERENCE_PACKAGE)
    enable_testing()
    add_test(NAME kitten.frontend-parity
        COMMAND "${KITTEN_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/check_parity.py"
            --binary $<TARGET_FILE:kitten-tts> --python-package "${KITTEN_REFERENCE_PACKAGE}")
    set_tests_properties(kitten.frontend-parity PROPERTIES TIMEOUT 600)
endif()

if(LLAMA_BUILD_TESTS)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME kitten.assets
            COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_assets.py" $<TARGET_FILE:kitten-tts>)
        set_tests_properties(kitten.assets PROPERTIES TIMEOUT 120)
    endif()
endif()

option(KITTEN_BUILD_PYTHON "Build the Python extension" OFF)
if(KITTEN_BUILD_PYTHON)
    get_target_property(http_definitions cpp-httplib INTERFACE_COMPILE_DEFINITIONS)
    if(NOT "CPPHTTPLIB_OPENSSL_SUPPORT" IN_LIST http_definitions)
        message(FATAL_ERROR "The Python package requires HTTPS. Install OpenSSL development headers or configure LLAMA_BUILD_BORINGSSL/LLAMA_BUILD_LIBRESSL.")
    endif()
    find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
    Python3_add_library(_native MODULE WITH_SOABI python.cpp)
    target_include_directories(_native PRIVATE ${TORCH_INCLUDE_DIRS})
    target_link_libraries(_native PRIVATE kitten-engine)
    set_target_properties(_native PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY "${KITTEN_PYTHON_OUTPUT_DIR}"
        RUNTIME_OUTPUT_DIRECTORY "${KITTEN_PYTHON_OUTPUT_DIR}")
    foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
        set_target_properties(_native PROPERTIES
            LIBRARY_OUTPUT_DIRECTORY_${config} "${KITTEN_PYTHON_OUTPUT_DIR}"
            RUNTIME_OUTPUT_DIRECTORY_${config} "${KITTEN_PYTHON_OUTPUT_DIR}")
    endforeach()
    if(APPLE)
        set_target_properties(_native PROPERTIES BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "@loader_path/../torch/lib")
    elseif(UNIX)
        target_link_options(_native PRIVATE "-Wl,--exclude-libs,ALL")
        set_target_properties(_native PROPERTIES BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN/../torch/lib")
    endif()
endif()
