option(KITTEN_CPU "Build the llama.cpp and LibTorch speech backend" ON)
option(KITTEN_CUDA "Build the native NVIDIA speech backend" OFF)
if(NOT KITTEN_CPU AND NOT KITTEN_CUDA)
    message(FATAL_ERROR "Enable KITTEN_CPU or KITTEN_CUDA")
endif()
if(KITTEN_CPU)
    find_package(Torch REQUIRED)
    set(KITTEN_BUILD_CLI OFF CACHE BOOL "Build normalizer CLI" FORCE)
    add_subdirectory(${PROJECT_SOURCE_DIR}/vendor/kitten-text-processing ${CMAKE_CURRENT_BINARY_DIR}/text-processing)
    add_library(kitten-text STATIC text.cpp)
    target_link_libraries(kitten-text PUBLIC KittenTextProcessing::normalizer)
    target_compile_features(kitten-text PUBLIC cxx_std_17)
    add_library(kitten-decoder STATIC decoder.cpp)
    target_include_directories(kitten-decoder PRIVATE ${PROJECT_SOURCE_DIR}/vendor/nlohmann)
    target_link_libraries(kitten-decoder PUBLIC ${TORCH_LIBRARIES})
    target_compile_features(kitten-decoder PUBLIC cxx_std_17)
endif()
if(KITTEN_CUDA)
    set(KT2N_CUDA ON CACHE STRING "Native CUDA backend" FORCE)
    set(KT2N_BUILD_TESTS OFF CACHE BOOL "Build native tests")
    set(KT2N_BUILD_TEXT ON CACHE BOOL "Build native normalization" FORCE)
    add_subdirectory(${PROJECT_SOURCE_DIR}/native ${CMAKE_CURRENT_BINARY_DIR}/native)
endif()
add_executable(kitten-tts main.cpp assets.cpp)
target_include_directories(kitten-tts PRIVATE ${PROJECT_SOURCE_DIR}/vendor/nlohmann)
target_compile_features(kitten-tts PRIVATE cxx_std_20)
target_compile_definitions(kitten-tts PRIVATE KITTEN_TTS_CPU=$<BOOL:${KITTEN_CPU}> KITTEN_TTS_NATIVE=$<BOOL:${KITTEN_CUDA}>)
target_link_libraries(kitten-tts PRIVATE llama-common)
if(KITTEN_CPU)
    target_link_libraries(kitten-tts PRIVATE llama kitten-text kitten-decoder)
endif()
if(KITTEN_CUDA)
    target_sources(kitten-tts PRIVATE native_cli.cpp)
    target_link_libraries(kitten-tts PRIVATE kt2n_engine kt2n_core)
    include(GNUInstallDirs)
    set_target_properties(kitten-tts PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
    # RUNPATH is not transitive: each shared dependency must find its siblings
    # when the unified CLI is installed outside the system library directories.
    foreach(kitten_dependency IN ITEMS llama-common llama ggml ggml-base ggml-cpu)
        if(TARGET ${kitten_dependency})
            set_property(TARGET ${kitten_dependency} APPEND PROPERTY INSTALL_RPATH "$ORIGIN")
        endif()
    endforeach()
endif()
install(TARGETS kitten-tts RUNTIME DESTINATION bin)
if(KITTEN_DATA_DIR)
    set(kitten_data "${KITTEN_DATA_DIR}")
else()
    set(kitten_data "${CMAKE_CURRENT_BINARY_DIR}/text-processing/data")
endif()
target_compile_definitions(kitten-tts PRIVATE KITTEN_DEFAULT_DATA="${kitten_data}")

set(KITTEN_REFERENCE_PACKAGE "" CACHE PATH "KittenTTS Python checkout for differential tests")
set(KITTEN_TEST_PYTHON "python3" CACHE FILEPATH "Python with KittenTTS2 conversion dependencies")
if(KITTEN_CPU AND BUILD_TESTING AND KITTEN_REFERENCE_PACKAGE)
    enable_testing()
    add_test(NAME kitten.frontend-parity
        COMMAND "${KITTEN_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/check_parity.py"
            --binary $<TARGET_FILE:kitten-tts> --python-package "${KITTEN_REFERENCE_PACKAGE}")
    set_tests_properties(kitten.frontend-parity PROPERTIES TIMEOUT 600)
endif()

if(LLAMA_BUILD_TESTS)
    add_executable(kitten-native-stats-test test_native_stats.cpp)
    target_include_directories(kitten-native-stats-test PRIVATE "${PROJECT_SOURCE_DIR}/native/include")
    target_compile_features(kitten-native-stats-test PRIVATE cxx_std_17)
    add_test(NAME kitten.native-stats COMMAND kitten-native-stats-test)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        if(KITTEN_CPU)
            add_test(NAME kitten.assets
                COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_assets.py" $<TARGET_FILE:kitten-tts>)
            set_tests_properties(kitten.assets PROPERTIES TIMEOUT 120)
        endif()
        if(KITTEN_CUDA)
            add_test(NAME kitten.native-assets
                COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_native_assets.py" $<TARGET_FILE:kitten-tts>)
            set_tests_properties(kitten.native-assets PROPERTIES TIMEOUT 120)
        endif()
    endif()
endif()
