if(NOT _kt2n_voice_av STREQUAL "OFF")
  set(_av_hint "")
  if(KT2N_VOICE_AV_SYSROOT)
    set(_av_hint "${KT2N_VOICE_AV_SYSROOT}/usr/include/x86_64-linux-gnu" "${KT2N_VOICE_AV_SYSROOT}/usr/include")
  endif()
  find_path(KT2N_AV_INCLUDE_DIR libavcodec/avcodec.h HINTS ${_av_hint})
  set(_av_libs "")
  foreach(_lib avformat avcodec swresample avutil)
    find_library(KT2N_AV_${_lib}_LIB NAMES ${_lib} lib${_lib}.so.62 lib${_lib}.so.61 lib${_lib}.so.60
                 lib${_lib}.so.59 lib${_lib}.so.58 lib${_lib}.so.56 lib${_lib}.so.5 lib${_lib}.so.4 lib${_lib}.so.3)
    list(APPEND _av_libs ${KT2N_AV_${_lib}_LIB})
  endforeach()
  if(KT2N_AV_INCLUDE_DIR AND KT2N_AV_avformat_LIB AND KT2N_AV_avcodec_LIB AND KT2N_AV_swresample_LIB AND KT2N_AV_avutil_LIB)
    add_library(kt2n_voice_decode STATIC decode.cc)
    target_include_directories(kt2n_voice_decode PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${KT2N_AV_INCLUDE_DIR}")
    target_link_libraries(kt2n_voice_decode PUBLIC ${_av_libs})
    target_compile_features(kt2n_voice_decode PUBLIC cxx_std_20)
    set_target_properties(kt2n_voice_decode PROPERTIES POSITION_INDEPENDENT_CODE ON)
    if(KT2N_HAVE_CUDA AND KT2N_BUILD_CLONING)
      # the clone build service behind engine.h kt2n_voice_build (decode + admission + LRU + GPU build)
      add_library(kt2n_voice_service STATIC voice_service.cc)
      target_link_libraries(kt2n_voice_service PUBLIC kt2n_voice kt2n_voice_decode kt2n_voice_policy kt2n_voice_blob
                        kt2n_voice_consent)
      target_include_directories(kt2n_voice_service PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
      set_target_properties(kt2n_voice_service PROPERTIES POSITION_INDEPENDENT_CODE ON)
    endif()
    message(STATUS "kt2n voice: in-process FFmpeg decode ON (${KT2N_AV_INCLUDE_DIR})")
  elseif(_kt2n_voice_av STREQUAL "ON" OR KT2N_BUILD_CLONING)
    message(FATAL_ERROR "Voice decoding requires the FFmpeg development headers/libraries (libavformat, libavcodec, libswresample, libavutil)")
  else()
    message(STATUS "kt2n voice: in-process FFmpeg decode OFF (headers or libraries not found)")
  endif()
endif()
