# L7 render state + render calls (include/kt2n/render/{rng,commit_plan,state,call,arena}.h).
#   kt2n_render       CPU: numpy-exact flow-noise RNG, commit geometry, paged pool, descriptors
#   kt2n_render_cuda  CUDA: device arena, L7 kernels (noise gather, prefix page copy), RenderCaller
add_library(kt2n_render STATIC rng.cc commit_plan.cc state.cc descriptors.cc)
add_library(kt2n::render ALIAS kt2n_render)
target_include_directories(kt2n_render PUBLIC "${PROJECT_SOURCE_DIR}/include")
target_compile_features(kt2n_render PUBLIC cxx_std_20)
target_link_libraries(kt2n_render PUBLIC kt2n_core)
kt2n_exact_numerics(kt2n_render)
kt2n_set_warnings(kt2n_render)


if(KT2N_HAVE_CUDA)
  add_library(kt2n_render_cuda STATIC render_call.cu service.cu)
  add_library(kt2n::render_cuda ALIAS kt2n_render_cuda)
  target_link_libraries(kt2n_render_cuda PUBLIC kt2n_render CUDA::cudart_static)  # no CUDA::cuda_driver: driver entry points resolve at run time (cu_driver.h)
  target_compile_definitions(kt2n_render_cuda PUBLIC KT2N_HAVE_CUDA=1)
  set_target_properties(kt2n_render_cuda PROPERTIES POSITION_INDEPENDENT_CODE ON)
  kt2n_cuda_target(kt2n_render_cuda)
  kt2n_set_warnings(kt2n_render_cuda)
  # The render program over the lanes' op libraries: L4 encoder (built here until its lane adds a
  # target), L5 estimator (kt2n_render_est from src/render/estimator).
  if(NOT TARGET kt2n_render_enc)
    add_library(kt2n_render_enc STATIC encoder/encoder.cu)
    target_include_directories(kt2n_render_enc PUBLIC "${PROJECT_SOURCE_DIR}/include" PRIVATE "${PROJECT_SOURCE_DIR}/src/core")
    target_link_libraries(kt2n_render_enc PUBLIC kt2n_core CUDA::cudart_static)
    set_target_properties(kt2n_render_enc PROPERTIES POSITION_INDEPENDENT_CODE ON)
    kt2n_cuda_target(kt2n_render_enc)
    target_compile_options(kt2n_render_enc PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
  endif()
  if(NOT TARGET kt2n_render_hift)
    add_library(kt2n_render_hift STATIC hift/hift.cu)
    target_include_directories(kt2n_render_hift PUBLIC "${PROJECT_SOURCE_DIR}/include")
    target_link_libraries(kt2n_render_hift PUBLIC CUDA::cudart_static)
    set_target_properties(kt2n_render_hift PROPERTIES POSITION_INDEPENDENT_CODE ON)
    kt2n_cuda_target(kt2n_render_hift)
    target_compile_options(kt2n_render_hift PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=-ffp-contract=off>)
  endif()
  if(TARGET kt2n_render_est)
    add_library(kt2n_render_flow STATIC program.cu hift_stage.cu renderer.cu)
    target_include_directories(kt2n_render_flow PRIVATE "${PROJECT_SOURCE_DIR}/src/core")
    target_link_libraries(kt2n_render_flow PUBLIC kt2n_render_cuda kt2n_render_enc kt2n_render_est kt2n_render_hift)
    set_target_properties(kt2n_render_flow PROPERTIES POSITION_INDEPENDENT_CODE ON)
    kt2n_cuda_target(kt2n_render_flow)
    target_compile_options(kt2n_render_flow PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
  endif()
endif()
